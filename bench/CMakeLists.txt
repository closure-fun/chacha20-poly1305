add_executable( chacha20_poly1305_bench bench.cxx )

target_link_libraries( chacha20_poly1305_bench PRIVATE chacha20_poly1305 )
target_compile_features( chacha20_poly1305_bench PRIVATE cxx_std_23 )

if( CMAKE_CONFIGURATION_TYPES )
	target_compile_definitions( chacha20_poly1305_bench PRIVATE CCP_BENCH_FLAGS="$<CONFIG>" )
elseif( CMAKE_BUILD_TYPE )
	target_compile_definitions( chacha20_poly1305_bench PRIVATE CCP_BENCH_FLAGS="${CMAKE_BUILD_TYPE}" )
else()
	target_compile_definitions( chacha20_poly1305_bench PRIVATE CCP_BENCH_FLAGS="unspecified" )
endif()

if( CHACHA20_POLY1305_BENCH_EXTERNAL )
	find_package( OpenSSL QUIET )
	if( OpenSSL_FOUND )
		target_link_libraries( chacha20_poly1305_bench PRIVATE OpenSSL::Crypto )
		target_compile_definitions( chacha20_poly1305_bench PRIVATE CCP_BENCH_OPENSSL )
		message( STATUS "chacha20_poly1305 bench: openssl ${OPENSSL_VERSION} found" )
	else()
		message( STATUS "chacha20_poly1305 bench: openssl not found, its column will be unavailable" )
	endif()

	find_path( SODIUM_INCLUDE_DIR sodium.h )
	find_library( SODIUM_LIBRARY NAMES sodium libsodium )
	if( SODIUM_INCLUDE_DIR AND SODIUM_LIBRARY )
		target_include_directories( chacha20_poly1305_bench PRIVATE "${SODIUM_INCLUDE_DIR}" )
		target_link_libraries( chacha20_poly1305_bench PRIVATE "${SODIUM_LIBRARY}" )
		target_compile_definitions( chacha20_poly1305_bench PRIVATE CCP_BENCH_SODIUM )
		message( STATUS "chacha20_poly1305 bench: libsodium found" )
	else()
		message( STATUS "chacha20_poly1305 bench: libsodium not found, its column will be unavailable" )
	endif()
endif()
